const UN_DIA = 60 * 60 * 24;

async function opinionesBooksy() {
  const respuesta = await fetch(salon.booksy, {
    next: { revalidate: UN_DIA }, // se actualiza sola cada día
  });
  const html = await respuesta.text();
  // Busca la nota: {"ratingValue":4.94,"reviewCount":405}
  const bloque = html.match(/"aggregateRating"\s*:\s*\{[^}]*\}/)?.[0];
  const nota = Number(bloque?.match(/"ratingValue"\s*:\s*"?([\d.]+)/)?.[1]);
  return { nota: formatearNota(nota), total };
}

export default function AntesDespues({ antes, despues, titulo }) {
  const [posicion, setPosicion] = useState(50);

  return (
    <div className="relative overflow-hidden">
      <Capa src={despues} alt="después" />
      {/* Antes: recortado hasta la barra */}
      <div style={{ clipPath: `inset(0 ${100 - posicion}% 0 0)` }}>
        <Capa src={antes} alt="antes" />
      </div>
      <input type="range" value={posicion}
        onChange={(e) => setPosicion(Number(e.target.value))} />
    </div>
  );
}

components/tienda/BotonRegalar.tsx
export default function BotonRegalar({ servicioId, nombre, precio }) {
  const { anadir } = useCarrito();
  const [anadido, setAnadido] = useState(false);

  if (anadido) return <Link href="/tienda/carrito">✓ Ver carrito</Link>;

  return (
    <button onClick={() => {
      anadir({ id: `bono-${servicioId}`, nombre: `Bono regalo: ${nombre}`, precio });
      setAnadido(true);
      setTimeout(() => setAnadido(false), 4000);
    }}>Regalar</button>
  );
}

Preguntas frecuentes

¿Cuánto tarda en estar lista mi web?

Una web esencial suele estar en 2 o 3 semanas. Una tienda online, entre 4 y 6, según el número de productos.

¿Podré cambiar cosas sin ayuda?

Sí. Te preparo un panel sencillo para editar textos, precios, fotos y el blog, y te enseño a usarlo.

Ya tengo Booksy o Instagram, ¿lo aprovechamos?

Claro. Conecto tus reservas, tus reseñas y tus redes para que la web y tus perfiles trabajen juntos.

¿Qué pasa después de publicarla?

Puedo encargarme del mantenimiento y de los cambios que necesites, o dejarte todo preparado para que lo gestiones tú.
